async function uploadFile() { const fileInput = document.getElementById('fileInput'); const file = fileInput.files[0]; if (!file) { alert("Please select a file first"); return; } const formData = new FormData(); formData.append('file', file); try { const res = await fetch('/api/upload', { method: 'POST', body: formData }); const data = await res.json(); addMessage(data.message, false); fileInput.value = ''; // reset } catch (e) { addMessage("❌ Upload failed", false); } }
